import sys, os, shutil, warnings
warnings.filterwarnings("ignore")
from anki.collection import Collection, ExportAnkiPackageOptions, DeckIdLimit
from anki.cards import FSRSMemoryState

out = sys.argv[1]
work = os.path.join(out, "col")
shutil.rmtree(work, ignore_errors=True); os.makedirs(work)
col = Collection(os.path.join(work, "collection.anki2"))
parent = col.decks.id("Испанский")
verbs = col.decks.id("Испанский::Глаголы")
basic_rev = col.models.by_name("Basic (and reversed card)")
basic = col.models.by_name("Basic")
cloze = col.models.by_name("Cloze")

# a media file
img = os.path.join(work, "gato.png")
with open(img, "wb") as f: f.write(b"\x89PNG\r\n\x1a\n" + b"fakepng" * 10)
snd = os.path.join(work, "hablar.mp3")
with open(snd, "wb") as f: f.write(b"ID3" + b"fakemp3" * 20)
img_name = col.media.add_file(img)
snd_name = col.media.add_file(snd)

def add(model, did, fields, tags=""):
    n = col.new_note(model)
    for k, v in fields.items(): n[k] = v
    n.tags = tags.split()
    col.add_note(n, did)
    return n

today = col.sched.today
# 1. reviewed with FSRS memory, both directions
n1 = add(basic_rev, verbs, {"Front": "<b>hablar</b>", "Back": "говорить&nbsp;<br>разговаривать"}, "verb")
c0, c1 = sorted(n1.cards(), key=lambda c: c.ord)
c0.type = 2; c0.queue = 2; c0.ivl = 12; c0.due = today + 5; c0.factor = 2500; c0.reps = 6; c0.lapses = 1
c0.memory_state = FSRSMemoryState(stability=14.25, difficulty=6.5)
col.update_card(c0)
c1.type = 2; c1.queue = 2; c1.ivl = 3; c1.due = today - 1; c1.factor = 2300; c1.reps = 2
c1.memory_state = FSRSMemoryState(stability=3.5, difficulty=7.25)
col.update_card(c1)
# 2. reviewed with SM-2 only, suspended
n2 = add(basic_rev, verbs, {"Front": "comer", "Back": "есть [sound:%s]" % snd_name})
c0 = sorted(n2.cards(), key=lambda c: c.ord)[0]
c0.type = 2; c0.queue = -1; c0.ivl = 30; c0.due = today + 20; c0.factor = 1300; c0.reps = 9; c0.lapses = 3
col.update_card(c0)
# 3. new, with an image in the parent deck
add(basic, parent, {"Front": "el gato <img src=\"%s\">" % img_name, "Back": "кот"})
# 4. new, entities
add(basic, parent, {"Front": "a &amp; b &lt;c&gt;", "Back": "&#1076;&#x430;"})
# 5. image only front -> can't be a pair
add(basic, parent, {"Front": "<img src=\"%s\">" % img_name, "Back": "кот"})
# 6. cloze
add(cloze, parent, {"Text": "Столица Испании — {{c1::Мадрид}}, Франции — {{c2::Париж}}"})
# 7. in learning (relearning, card due in seconds)
n7 = add(basic, verbs, {"Front": "vivir", "Back": "жить"})
c = n7.cards()[0]
import time
c.type = 3; c.queue = 1; c.ivl = 2; c.due = int(time.time()) + 600; c.factor = 2500; c.reps = 4; c.lapses = 1; c.left = 1
col.update_card(c)
col.db.execute("insert into revlog (id,cid,usn,ease,ivl,lastIvl,factor,time,type) values (?,?,?,?,?,?,?,?,?)",
               int(time.time()*1000) - 3*86400000, sorted(n1.cards(), key=lambda c: c.ord)[0].id, -1, 3, 12, 4, 2500, 5000, 1)

for legacy in (False, True):
    name = "legacy.apkg" if legacy else "modern.apkg"
    col.export_anki_package(out_path=os.path.join(out, name),
        options=ExportAnkiPackageOptions(with_scheduling=True, with_deck_configs=True, with_media=True, legacy=legacy),
        limit=DeckIdLimit(parent))
print("crt", col.crt, "today", today)
col.close()
